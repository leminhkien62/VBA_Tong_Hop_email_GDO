'==========================================================
' HAM DUNG CHUOI TIENG VIET AN TOAN (so = ma Unicode)
'==========================================================
Function VN(ParamArray Parts() As Variant) As String

    Dim k As Long
    Dim sResult As String

    For k = LBound(Parts) To UBound(Parts)
        If VarType(Parts(k)) = vbString Then
            sResult = sResult & CStr(Parts(k))
        Else
            sResult = sResult & UniChr(CLng(Parts(k)))
        End If
    Next k

    VN = sResult

End Function

'----------------------------------------------------------
' Dung chu Han tu danh sach ma hex, vd: CJK("6642 9593")
'----------------------------------------------------------
Function CJK(ByVal HexList As String) As String

    Dim a() As String
    Dim k As Long
    Dim s As String

    a = Split(HexList, " ")
    For k = LBound(a) To UBound(a)
        If Len(a(k)) > 0 Then s = s & UniChr(HexToLong(a(k)))
    Next k

    CJK = s

End Function

'----------------------------------------------------------
' Doi chuoi hex -> Long
'----------------------------------------------------------
Function HexToLong(ByVal h As String) As Long

    Dim i As Long
    Dim v As Long
    Dim d As Long

    For i = 1 To Len(h)
        d = InStr("0123456789ABCDEF", UCase$(Mid$(h, i, 1))) - 1
        If d < 0 Then d = 0
        v = v * 16 + d
    Next i

    HexToLong = v

End Function

'----------------------------------------------------------
' ChrW an toan cho ma > 32767
'----------------------------------------------------------
Function UniChr(ByVal code As Long) As String

    If code > 32767 Then code = code - 65536
    UniChr = ChrW(code)

End Function

'==========================================================
' HIEN FORM NHAP LIEU (frmNhapLieu1): email, ngay, ngon ngu
' bFull = True : hien du email + tu ngay + den ngay + ngon ngu
' bFull = False: chi hien ngon ngu dich
' Tra ve False neu bam Huy
'==========================================================
Function HienForm(ByVal bFull As Boolean, ByRef sSender As String, _
                  ByRef dFrom As Date, ByRef dTo As Date, ByRef sLang As String) As Boolean

    Dim f As Object

    On Error GoTo NoForm
    Set f = VBA.UserForms.Add("frmNhapLieu1")
    On Error GoTo 0

    f.Setup bFull
    f.Show vbModal

    If CStr(f.Tag) = "OK" Then
        If bFull Then
            sSender = CStr(f.SenderText)
            dFrom = f.FromValue
            dTo = f.ToValue
        End If
        sLang = CStr(f.LangCode)
        HienForm = True
    End If

    Unload f
    Exit Function

NoForm:
    MsgBox "Chua co form frmNhapLieu1." & vbCrLf & _
           "Vao File > Import File..., chon frmNhapLieu1.frm", vbExclamation

End Function

'==========================================================
' MAIN: quet de quy Inbox + tat ca thu muc con
'==========================================================
Sub TongHopEmailOutlook()

    Dim olNs As Outlook.NameSpace
    Dim Inbox As Outlook.MAPIFolder

    Dim xlApp As Object
    Dim wb As Object
    Dim ws As Object

    Dim SenderInput As String
    Dim arrSenders() As String
    Dim FromDate As Date
    Dim ToDate As Date

    Dim Row As Long
    Dim CountMail As Long

    Dim BaseFolder As String
    Dim MsgFolder As String
    Dim XlsxPath As String

    If Not HienForm(True, SenderInput, FromDate, ToDate, gLangTo) Then Exit Sub

    arrSenders = TachDanhSachEmail(SenderInput)

    On Error GoTo ErrorHandler

    BaseFolder = Environ$("USERPROFILE") & "\Desktop\" & "H " & Format(Now, "yyyy-mm-dd_HH-mm") & "_TongHopEmail"
    MsgFolder = BaseFolder & "\H Email"

    CreateFolderIfNotExist BaseFolder
    CreateFolderIfNotExist MsgFolder

    Set olNs = Application.GetNamespace("MAPI")
    Set Inbox = olNs.GetDefaultFolder(olFolderInbox)

    Set xlApp = CreateObject("Excel.Application")
    xlApp.Visible = True

    Set wb = xlApp.Workbooks.Add
    Set ws = wb.Worksheets(1)
    ws.Name = "TongHopEmail"

    KhoiTaoSheet ws

    Row = 3            ' dong 1 = tieu de, dong 2 = header, du lieu tu dong 3
    CountMail = 0

    XuLyThuMucDeQuy Inbox, arrSenders, FromDate, ToDate, ws, Row, CountMail, MsgFolder

    DinhDangExcel ws, Row - 1

    XlsxPath = BaseFolder & "\H " & Format(Now, "yyyy-mm-dd_HH-mm-ss") & "_TongHopEmail.xlsx"
    wb.SaveAs XlsxPath

    MsgBox _
        "Da hoan thanh!" & vbCrLf & vbCrLf & _
        "So email: " & CountMail & vbCrLf & vbCrLf & _
        "Excel:" & vbCrLf & XlsxPath & vbCrLf & vbCrLf & _
        "Email (.msg):" & vbCrLf & MsgFolder, _
        vbInformation, _
        "Tong hop email"

    Exit Sub

ErrorHandler:

    MsgBox _
        "Co loi xay ra:" & vbCrLf & vbCrLf & _
        Err.Number & " - " & Err.Description & vbCrLf & vbCrLf & _
        "Buoc: " & gStep, _
        vbCritical, _
        "Loi"

End Sub

'==========================================================
' DE QUY QUET 1 THU MUC + TAT CA THU MUC CON
'==========================================================
Sub XuLyThuMucDeQuy( _
    ByVal Fld As Outlook.MAPIFolder, _
    ByRef arrSenders() As String, _
    ByVal FromDate As Date, _
    ByVal ToDate As Date, _
    ByVal ws As Object, _
    ByRef Row As Long, _
    ByRef CountMail As Long, _
    ByVal MsgFolder As String)

    Dim Items As Outlook.Items
    Dim Mail As Outlook.MailItem
    Dim SubFld As Outlook.MAPIFolder
    Dim i As Long

    On Error Resume Next
    Set Items = Fld.Items
    On Error GoTo 0

    If Items Is Nothing Then Exit Sub

    On Error Resume Next
    Items.Sort "[ReceivedTime]", True
    On Error GoTo 0

    For i = 1 To Items.Count

        gStep = "quet thu muc / doc email"

        If TypeName(Items(i)) = "MailItem" Then

            Set Mail = Items(i)

            If Mail.ReceivedTime >= FromDate _
               And Mail.ReceivedTime < DateAdd("d", 1, ToDate) Then

                If CoTrongDanhSach(GetSenderEmail(Mail), arrSenders) _
                   And CoTieuDeHopLe(Mail.Subject) Then

                    CountMail = CountMail + 1

                    ProcessMail Mail, ws, Row, MsgFolder

                    Row = Row + 1

                End If

            End If

        End If

        If i Mod 100 = 0 Then DoEvents

    Next i

    For Each SubFld In Fld.Folders
        XuLyThuMucDeQuy SubFld, arrSenders, FromDate, ToDate, ws, Row, CountMail, MsgFolder
    Next SubFld

End Sub

'==========================================================
' MACRO 2: TONG HOP EXCEL TU CAC FILE .MSG DA TAI SAN
'==========================================================
Sub TongHopExcelTuFileMsg()

    Dim olNs As Outlook.NameSpace
    Dim Mail As Outlook.MailItem

    Dim xlApp As Object
    Dim wb As Object
    Dim ws As Object

    Dim FSO As Object
    Dim FolderObj As Object
    Dim FileItem As Object

    Dim FolderPath As String
    Dim BaseFolder As String
    Dim XlsxPath As String

    Dim Row As Long
    Dim CountMail As Long

    FolderPath = InputBox( _
        "Nhap duong dan thu muc chua cac file email (.msg) da tai:", _
        "Chon thu muc", _
        Environ$("USERPROFILE") & "\Desktop\TongHopEmail\Email")

    If Trim(FolderPath) = "" Then Exit Sub

    If Right$(FolderPath, 1) = "\" Then
        FolderPath = Left$(FolderPath, Len(FolderPath) - 1)
    End If

    Set FSO = CreateObject("Scripting.FileSystemObject")

    If Not FSO.FolderExists(FolderPath) Then
        MsgBox "Khong tim thay thu muc nay.", vbExclamation
        Exit Sub
    End If

    Dim sDummy As String
    Dim dDummy1 As Date
    Dim dDummy2 As Date

    If Not HienForm(False, sDummy, dDummy1, dDummy2, gLangTo) Then Exit Sub

    Set FolderObj = FSO.GetFolder(FolderPath)
    Set olNs = Application.GetNamespace("MAPI")

    Set xlApp = CreateObject("Excel.Application")
    xlApp.Visible = True

    Set wb = xlApp.Workbooks.Add
    Set ws = wb.Worksheets(1)
    ws.Name = "TongHopEmail"

    KhoiTaoSheet ws

    Row = 3
    CountMail = 0

    For Each FileItem In FolderObj.Files

        If LCase$(FSO.GetExtensionName(FileItem.Path)) = "msg" Then

            On Error Resume Next
            Set Mail = Nothing
            Set Mail = olNs.OpenSharedItem(FileItem.Path)
            On Error GoTo 0

            If Not Mail Is Nothing Then

                If CoTieuDeHopLe(Mail.Subject) Then

                    CountMail = CountMail + 1

                    GhiDongEmail Mail, ws, Row

                    If Mail.Attachments.Count > 0 Then
                        ws.Hyperlinks.Add _
                            Anchor:=ws.Cells(Row, 6), _
                            Address:=FileItem.Path, _
                            TextToDisplay:=VN("M", 7903, " email")
                    End If

                    Row = Row + 1

                End If

                On Error Resume Next
                Mail.Close olDiscard
                On Error GoTo 0
                Set Mail = Nothing

            End If

        End If

    Next FileItem

    DinhDangExcel ws, Row - 1

    BaseFolder = FSO.GetParentFolderName(FolderPath)
    If BaseFolder = "" Then BaseFolder = FolderPath

    XlsxPath = BaseFolder & "\TongHopEmail.xlsx"
    wb.SaveAs XlsxPath

    MsgBox _
        "Da hoan thanh!" & vbCrLf & vbCrLf & _
        "So email: " & CountMail & vbCrLf & vbCrLf & _
        "Excel:" & vbCrLf & XlsxPath, _
        vbInformation, _
        "Tong hop email"

End Sub

'==========================================================
' KIEM TRA SUBJECT
'==========================================================
Function CoTieuDeHopLe(ByVal Subject As String) As Boolean

    Dim sSubj As String
    Dim sMarker As String

    sSubj = ChuanHoa(Subject)

    If Len(sSubj) > 0 Then
        If Left$(sSubj, 1) = "[" Then
            sSubj = Mid$(sSubj, 2)
        End If
    End If

    sMarker = TieuDeYeuCau()

    CoTieuDeHopLe = (InStr(1, sSubj, sMarker, vbTextCompare) = 1)

End Function

'==========================================================
' DINH DANG COT / VIEN / AUTOFILTER / ZOOM 80
'==========================================================
Sub DinhDangExcel(ByVal ws As Object, ByVal LastRow As Long)

    If LastRow < 3 Then LastRow = 3

    With ws

        .Columns("A:A").ColumnWidth = 6
        .Columns("B:B").ColumnWidth = 20
        .Columns("C:C").ColumnWidth = 55
        .Columns("D:D").ColumnWidth = 55
        .Columns("E:E").ColumnWidth = 28
        .Columns("F:F").ColumnWidth = 14
        .Columns("G:G").ColumnWidth = 20

        .Range("B3:B" & LastRow).NumberFormat = "dd/mm/yyyy hh:mm:ss"

        With .Range("A3:G" & LastRow)
            .VerticalAlignment = -4160        ' top
            .WrapText = True
        End With

        .Range("A3:B" & LastRow).HorizontalAlignment = -4108
        .Range("F3:F" & LastRow).HorizontalAlignment = -4108

        With .Range("A1:G" & LastRow).Borders
            .LineStyle = 1
            .Weight = 2
        End With

        .Rows("3:" & LastRow).AutoFit

        If .AutoFilterMode Then .AutoFilterMode = False
        .Range("A2:G" & LastRow).AutoFilter

        .Activate
        .Application.ActiveWindow.Zoom = 80
        .Range("A3").Select
        .Application.ActiveWindow.FreezePanes = False
        .Application.ActiveWindow.FreezePanes = True

    End With

End Sub

'==========================================================
' GHI 1 DONG THONG TIN EMAIL VAO EXCEL (+ DICH)
'==========================================================
Sub GhiDongEmail(ByVal Mail As Outlook.MailItem, ByVal ws As Object, ByVal Row As Long)

    Dim sBody As String
    Dim sNoiDung As String

    gStep = "doc Body / tach noi dung"
    sBody = LayBody(Mail)
    sNoiDung = LayNoiDungPhanAnh(sBody)

    gStep = "ghi STT, thoi gian"
    ws.Cells(Row, 1).Value = Row - 2
    ws.Cells(Row, 2).Value = Mail.ReceivedTime

    gStep = "ghi noi dung (cot C)"
    ws.Cells(Row, 3).Value = sNoiDung

    If gLangTo <> "0" And gLangTo <> "" And Len(sNoiDung) > 0 Then
        gStep = "dich bang Ollama (cot D)"
        ws.Cells(Row, 4).Value = CStr(Dich(sNoiDung, gLangTo))
    End If

    gStep = "tach thong tin lien lac (cot E)"
    ws.Cells(Row, 5).Value = LayThongTinLienLac(sBody)

End Sub

'==========================================================
' XU LY 1 EMAIL (ghi dong + luu .msg neu co file dinh kem)
'==========================================================
Sub ProcessMail( _
    ByVal Mail As Outlook.MailItem, _
    ByVal ws As Object, _
    ByVal Row As Long, _
    ByVal MsgFolder As String)

    Dim SafeName As String
    Dim MsgPath As String
    Dim ErrNum As Long
    Dim ErrDesc As String

    GhiDongEmail Mail, ws, Row

    If Mail.Attachments.Count = 0 Then Exit Sub

    SafeName = CleanFileName( _
        "H " & Format(Mail.ReceivedTime, "yyyymmdd_hhnnss") _
        & "_" & Mail.Subject)

    If Len(SafeName) > 150 Then SafeName = Left$(SafeName, 150)

    MsgPath = MsgFolder & "\" & SafeName & ".msg"

    On Error Resume Next
    Err.Clear
    Mail.SaveAs MsgPath, olMSG
    ErrNum = Err.Number
    ErrDesc = Err.Description
    On Error GoTo 0

    If ErrNum <> 0 Then
        ws.Cells(Row, 6).Value = VN("L", 7895, "i l", 432, "u: ") & ErrNum & " - " & ErrDesc
    Else
        ws.Hyperlinks.Add _
            Anchor:=ws.Cells(Row, 6), _
            Address:=MsgPath, _
            TextToDisplay:=VN("M", 7903, " email")
    End If

End Sub

'==========================================================
' LAY BODY: neu Body rong thi lay tu HTMLBody (bo the HTML)
'==========================================================
Function LayBody(ByVal Mail As Outlook.MailItem) As String

    Dim s As String
    Dim p1 As Long
    Dim p2 As Long

    On Error Resume Next
    s = CStr(Mail.Body)
    On Error GoTo 0

    If Len(Trim$(s)) > 0 Then
        LayBody = s
        Exit Function
    End If

    On Error Resume Next
    s = CStr(Mail.HTMLBody)
    On Error GoTo 0

    s = Replace(s, "</td>", " ", , , vbTextCompare)
    s = Replace(s, "</tr>", " ", , , vbTextCompare)
    s = Replace(s, "<br>", " ", , , vbTextCompare)
    s = Replace(s, "<br/>", " ", , , vbTextCompare)
    s = Replace(s, "<br />", " ", , , vbTextCompare)
    s = Replace(s, "</p>", " ", , , vbTextCompare)

    Do
        p1 = InStr(1, s, "<", vbBinaryCompare)
        If p1 = 0 Then Exit Do
        p2 = InStr(p1, s, ">", vbBinaryCompare)
        If p2 = 0 Then Exit Do
        s = Left$(s, p1 - 1) & " " & Mid$(s, p2 + 1)
    Loop

    s = Replace(s, "&nbsp;", " ")
    s = Replace(s, "&amp;", "&")
    s = Replace(s, "&lt;", "<")
    s = Replace(s, "&gt;", ">")
    s = Replace(s, "&quot;", """")

    LayBody = s

End Function

'==========================================================
' TIM VI TRI GAN NHAT TRONG SO NHIEU MOC KET THUC
'==========================================================
Function ViTriGanNhat(ByVal Body As String, ByVal FromPos As Long, ByRef BestPos As Long, ByVal sMarker As String, Optional ByVal Compare As VbCompareMethod = vbBinaryCompare) As Long

    Dim p As Long
    p = InStr(FromPos, Body, sMarker, Compare)

    If p > 0 Then
        If BestPos = 0 Or p < BestPos Then
            BestPos = p
        End If
    End If

    ViTriGanNhat = BestPos

End Function

'==========================================================
' CHUAN HOA CHUOI: NBSP, xuong dong, tab -> 1 dau cach
'==========================================================
Function ChuanHoa(ByVal s As String) As String

    s = Replace(s, ChrW(160), " ")      ' NBSP
    s = Replace(s, ChrW(8203), "")      ' zero-width space
    s = Replace(s, ChrW(&HFE0F), "")    ' emoji variation selector
    s = Replace(s, vbTab, " ")
    s = Replace(s, vbCrLf, " ")
    s = Replace(s, vbCr, " ")
    s = Replace(s, vbLf, " ")

    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop

    ChuanHoa = Trim$(s)

End Function

'==========================================================
' RUT GON NOI DUNG PHAN ANH / GOP Y TU BODY EMAIL
'==========================================================
Function LayNoiDungPhanAnh(ByVal Body As String) As String

    Dim sMarker As String, sShort As String, sLienLac As String
    Dim sPhone As String, sGlobe As String, sWarn As String
    Dim startPos As Long, endPos As Long, mLen As Long
    Dim sResult As String

    Body = ChuanHoa(Body)

    sMarker = VN("N", 7897, "i dung ph", 7843, "n ", 225, "nh / G", 243, "p ", 253, ":")
    sShort = VN("G", 243, "p ", 253, ":")          ' du phong: chi tim "Gop y:"

    startPos = InStr(1, Body, sMarker, vbTextCompare)
    mLen = Len(sMarker)

    If startPos = 0 Then
        startPos = InStr(1, Body, sShort, vbTextCompare)
        mLen = Len(sShort)
    End If

    If startPos = 0 Then
        LayNoiDungPhanAnh = Body
        Exit Function
    End If

    startPos = startPos + mLen

    sLienLac = NhanLienLac() & ":"
    sPhone = ChrW(&HD83D) & ChrW(&HDCDE)        ' U+1F4DE
    sGlobe = ChrW(&HD83C) & ChrW(&HDF0F)        ' U+1F30F
    sWarn = ChrW(&H26A0)

    endPos = 0
    ViTriGanNhat Body, startPos, endPos, sLienLac, vbTextCompare
    ViTriGanNhat Body, startPos, endPos, sPhone, vbBinaryCompare
    ViTriGanNhat Body, startPos, endPos, sGlobe, vbBinaryCompare
    ViTriGanNhat Body, startPos, endPos, sWarn, vbBinaryCompare

    If endPos = 0 Then endPos = Len(Body) + 1

    sResult = Mid$(Body, startPos, endPos - startPos)
    sResult = Replace(sResult, sPhone, "")
    sResult = Replace(sResult, sGlobe, "")
    sResult = Replace(sResult, sWarn, "")

    LayNoiDungPhanAnh = Trim$(sResult)

End Function

'==========================================================
' RUT GON THONG TIN LIEN LAC TU BODY EMAIL
'==========================================================
Function LayThongTinLienLac(ByVal Body As String) As String

    Dim sMarker As String
    Dim sGlobe As String
    Dim sWarn As String
    Dim startPos As Long
    Dim endPos As Long
    Dim sResult As String

    Body = ChuanHoa(Body)

    sMarker = NhanLienLac() & ":"

    startPos = InStr(1, Body, sMarker, vbTextCompare)

    If startPos = 0 Then
        LayThongTinLienLac = ""
        Exit Function
    End If

    startPos = startPos + Len(sMarker)

    sGlobe = ChrW(&HD83C) & ChrW(&HDF0F)
    sWarn = ChrW(&H26A0)

    endPos = 0
    ViTriGanNhat Body, startPos, endPos, sGlobe, vbBinaryCompare
    ViTriGanNhat Body, startPos, endPos, sWarn, vbBinaryCompare

    If endPos = 0 Then endPos = Len(Body) + 1

    sResult = Mid$(Body, startPos, endPos - startPos)

    LayThongTinLienLac = Trim$(sResult)

End Function

'==========================================================
' TACH DANH SACH EMAIL NGUOI GUI
'==========================================================
Function TachDanhSachEmail(ByVal sInput As String) As String()

    Dim sTemp As String
    Dim arr() As String
    Dim k As Long

    sTemp = Replace(sInput, ";", ",")
    arr = Split(sTemp, ",")

    For k = LBound(arr) To UBound(arr)
        arr(k) = LCase(Trim$(arr(k)))
    Next k

    TachDanhSachEmail = arr

End Function

Function CoTrongDanhSach(ByVal Email As String, ByRef arrList() As String) As Boolean

    Dim k As Long
    Dim sEmail As String

    sEmail = LCase(Trim$(Email))

    For k = LBound(arrList) To UBound(arrList)
        If arrList(k) <> "" And sEmail = arrList(k) Then
            CoTrongDanhSach = True
            Exit Function
        End If
    Next k

    CoTrongDanhSach = False

End Function

'==========================================================
' LAY EMAIL NGUOI GUI
'==========================================================
Function GetSenderEmail(ByVal Mail As Outlook.MailItem) As String

    On Error Resume Next

    Dim Sender As Outlook.AddressEntry
    Dim ExUser As Outlook.ExchangeUser

    If Mail.SenderEmailType = "EX" Then

        Set Sender = Mail.Sender

        If Not Sender Is Nothing Then

            Set ExUser = Sender.GetExchangeUser

            If Not ExUser Is Nothing Then
                GetSenderEmail = ExUser.PrimarySmtpAddress
            Else
                GetSenderEmail = Mail.SenderEmailAddress
            End If

        Else
            GetSenderEmail = Mail.SenderEmailAddress
        End If

    Else
        GetSenderEmail = Mail.SenderEmailAddress
    End If

    On Error GoTo 0

End Function

'==========================================================
' TAO FOLDER / XU LY TEN FILE
'==========================================================
Sub CreateFolderIfNotExist(ByVal FolderPath As String)

    If Dir(FolderPath, vbDirectory) = "" Then
        MkDir FolderPath
    End If

End Sub

Function CleanFileName(ByVal FileName As String) As String

    Dim BadChars As Variant
    Dim c As Variant

    BadChars = Array("\", "/", ":", "*", "?", """", "<", ">", "|")

    For Each c In BadChars
        FileName = Replace(FileName, CStr(c), "_")
    Next c

    FileName = Trim$(FileName)

    If Len(FileName) = 0 Then FileName = "Email"

    CleanFileName = FileName

End Function

'==========================================================
'  PHAN DICH BANG OLLAMA (translategemma:12b)
'==========================================================
Function Dich(ByVal text As String, Optional ByVal langTo As String = "vi") As String

    Dim http As Object
    Dim url As String
    Dim json As String
    Dim prompt As String

    If Len(Trim$(text)) = 0 Then
        Dich = ""
        Exit Function
    End If

    url = "http://192.168.40.189:11434/api/generate"

    prompt = "Translate the following text to " & GetLanguageName(langTo) & _
             ". Only return the translated text. Do not explain anything." & vbLf & _
             vbLf & text

    json = "{""model"":""translategemma:12b""," & _
           """prompt"":""" & JsonEscape(prompt) & """," & _
           """stream"":false}"

    On Error GoTo ErrorHandler

    Set http = CreateObject("MSXML2.ServerXMLHTTP.6.0")
    http.setTimeouts 5000, 5000, 30000, 180000     ' resolve, connect, send, receive (ms)

    http.Open "POST", url, False
    http.setRequestHeader "Content-Type", "application/json; charset=utf-8"
    http.Send StringToUtf8(json)

    If http.Status <> 200 Then
        Dich = "[Loi Ollama: HTTP " & http.Status & "]"
        Exit Function
    End If

    Dich = Trim$(GetJsonValue(Utf8ToString(http.responseBody), "response"))

    Exit Function

ErrorHandler:
    Dich = "[Loi: " & Err.Description & "]"
End Function

'----------------------------------------------------------
' Ma ngon ngu -> ten ngon ngu
'----------------------------------------------------------
Function GetLanguageName(ByVal lang As String) As String

    Select Case LCase(lang)
        Case "vi":    GetLanguageName = "Vietnamese"
        Case "en":    GetLanguageName = "English"
        Case "zh":    GetLanguageName = "Simplified Chinese"
        Case "zh-tw": GetLanguageName = "Traditional Chinese"
        Case "ja":    GetLanguageName = "Japanese"
        Case "ko":    GetLanguageName = "Korean"
        Case "de":    GetLanguageName = "German"
        Case "fr":    GetLanguageName = "French"
        Case Else:    GetLanguageName = lang
    End Select

End Function

'----------------------------------------------------------
' Escape chuoi de dua vao JSON
'----------------------------------------------------------
Function JsonEscape(ByVal s As String) As String

    s = Replace(s, "\", "\\")
    s = Replace(s, """", "\""")
    s = Replace(s, vbCrLf, "\n")
    s = Replace(s, vbCr, "\n")
    s = Replace(s, vbLf, "\n")
    s = Replace(s, vbTab, "\t")

    JsonEscape = s

End Function

'----------------------------------------------------------
' Lay gia tri chuoi cua key trong JSON (giai ma \n \" \\ \uXXXX)
'----------------------------------------------------------
Function GetJsonValue(ByVal json As String, ByVal key As String) As String

    Dim p As Long
    Dim ch As String
    Dim nx As String
    Dim sb As String

    p = InStr(json, """" & key & """:""")
    If p = 0 Then
        GetJsonValue = ""
        Exit Function
    End If

    p = p + Len(key) + 4

    Do While p <= Len(json)

        ch = Mid$(json, p, 1)

        If ch = """" Then Exit Do

        If ch = "\" Then
            p = p + 1
            nx = Mid$(json, p, 1)
            Select Case nx
                Case "n": sb = sb & vbCrLf
                Case "r": ' bo qua
                Case "t": sb = sb & vbTab
                Case "b", "f": ' bo qua
                Case "u"
                    sb = sb & UniChr(HexToLong(Mid$(json, p + 1, 4)))
                    p = p + 4
                Case Else: sb = sb & nx          ' \" \\ \/
            End Select
        Else
            sb = sb & ch
        End If

        p = p + 1

    Loop

    GetJsonValue = sb

End Function

'----------------------------------------------------------
' Chuyen chuoi <-> mang byte UTF-8
'----------------------------------------------------------
Function StringToUtf8(ByVal s As String) As Variant

    Dim st As Object
    Set st = CreateObject("ADODB.Stream")

    st.Type = 2                 ' text
    st.Charset = "utf-8"
    st.Open
    st.WriteText s
    st.Position = 0
    st.Type = 1                 ' binary
    st.Position = 3             ' bo BOM
    StringToUtf8 = st.Read
    st.Close

End Function

Function Utf8ToString(ByVal b As Variant) As String

    Dim st As Object
    Set st = CreateObject("ADODB.Stream")

    st.Type = 1                 ' binary
    st.Open
    st.Write b
    st.Position = 0
    st.Type = 2                 ' text
    st.Charset = "utf-8"
    Utf8ToString = st.ReadText
    st.Close

End Function
